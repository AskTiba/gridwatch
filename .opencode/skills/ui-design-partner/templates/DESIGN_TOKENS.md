## Color Tokens (CSS Custom Properties)

```css
:root {
  /* ── Brand palette ──────────────────────────── */
  --color-primary:          /* Main brand color (e.g. hsl(220, 90%, 50%)) */;
  --color-primary-hover:    /* 10% darker for hover states */;
  --color-primary-contrast: /* Text on primary background (usually white) */;

  --color-secondary:        /* Secondary accent */;
  --color-secondary-hover:  /* */;

  /* ── Neutral surfaces (light theme) ────────── */
  --bg:                     /* Page background — near-white with brand hint */;
  --surface:                /* Card / panel background */;
  --surface-elevated:       /* Elevated card / modal background */;
  --border:                 /* Divider / border color */;

  /* ── Text (light theme) ─────────────────────── */
  --text:                   /* Primary text — ≥4.5:1 contrast on --bg */;
  --text-muted:             /* Secondary text */;
  --text-subtle:            /* Placeholder / disabled text */;

  /* ── Functional set (semantic) ──────────────── */
  --color-success:          /* Green — gain, positive, done */;
  --color-danger:           /* Red — error, destructive */;
  --color-warning:          /* Amber — caution, alert */;
  --color-info:             /* Blue — informational */;

  /* ── Motion tokens ───────────────────────────── */
  --ease-out:               cubic-bezier(0.22, 1, 0.36, 1);
  --ease-in-out:            cubic-bezier(0.4, 0, 0.2, 1);
  --dur-fast:               150ms;
  --dur-normal:             300ms;
  --dur-slow:               600ms;

  /* ── Spacing scale (4/8pt grid) ─────────────── */
  --space-1:  4px;
  --space-2:  8px;
  --space-3:  12px;
  --space-4:  16px;
  --space-6:  24px;
  --space-8:  32px;
  --space-12: 48px;
  --space-16: 64px;

  /* ── Radius ──────────────────────────────────── */
  --radius-sm:   6px;
  --radius-md:   12px;
  --radius-lg:   16px;
  --radius-xl:   24px;
  --radius-full: 9999px;
}

/* ── Dark theme ──────────────────────────────── */
@media (prefers-color-scheme: dark) {
  :root {
    --bg:             /* Near-black with brand hint */;
    --surface:        /* Slightly lighter than --bg */;
    --surface-elevated: /* Card elevated above surface */;
    --border:         /* Subtle divider — lighter than surface */;
    --text:           /* Near-white — ≥4.5:1 on --bg */;
    --text-muted:     /* */;
    --text-subtle:    /* */;

    /* Primary adjusted for dark mode contrast */
    --color-primary:  /* Brighter variant of same hue */;
  }
}
```

## Typography Scale

```css
/* Fluid type scale using clamp() */
:root {
  --text-xs:   clamp(0.75rem,  0.7rem + 0.25vw,  0.875rem);
  --text-sm:   clamp(0.875rem, 0.8rem + 0.375vw, 1rem);
  --text-base: clamp(1rem,     0.9rem + 0.5vw,   1.125rem);  /* ≥16px always */
  --text-lg:   clamp(1.125rem, 1rem + 0.625vw,   1.25rem);
  --text-xl:   clamp(1.25rem,  1.1rem + 0.75vw,  1.5rem);
  --text-2xl:  clamp(1.5rem,   1.3rem + 1vw,     2rem);
  --text-3xl:  clamp(1.875rem, 1.5rem + 1.875vw, 2.5rem);
  --text-4xl:  clamp(2.25rem,  1.75rem + 2.5vw,  3.5rem);
  --text-hero: clamp(2.5rem,   2rem + 3vw,       4.5rem);

  --font-body:    'Inter', system-ui, sans-serif;
  --font-heading: 'Inter', system-ui, sans-serif; /* Change if different */

  --leading-tight:  1.25;
  --leading-normal: 1.5;
  --leading-relaxed: 1.7;
}
```
